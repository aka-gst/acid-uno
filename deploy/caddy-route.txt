# Маршрут для Caddy. Вставляется в блок сайта ПЕРЕД `root * /srv`.
#
# Статика игры отдаётся файлово из каталога сайта — на контейнер уходят
# только запросы комнаты. Префикс /acid срезается: сервер ждёт /api/...,
# ровно как при локальном запуске.
#
# flush_interval -1 обязателен. Состояние стола идёт потоком SSE, и с
# буферизацией по умолчанию он копится в прокси: у игроков просто
# замирает стол, без единой ошибки в журнале.

handle /acid/api/* {
  uri strip_prefix /acid
  reverse_proxy acid-uno:4173 {
    flush_interval -1
  }
}
